31def growth_policy(peak_photo_count): 

32 if type(peak_photo_count) is not int or peak_photo_count < 0: 

33 raise LayoutValidationError("사진 수는 0 이상의 정수여야 합니다.") 

34 level = max(i for i, row in enumerate(GROWTH) if peak_photo_count >= row[0]) 

35 minimum, name, width, depth, items, frames = GROWTH[level] 

36 return {"level": level, "min_photos": minimum, "name": name, 

37 "room": {"width": width, "depth": depth, "height": 3.4}, 

38 "limits": {"items": items, "frames": frames}}

54def _number(value, label): 

55 try: 

56 valid = type(value) in (int, float) and isfinite(value) 

57 except OverflowError: 

58 valid = False 

59 if not valid: 

60 raise LayoutValidationError(f"{label} 값은 유한한 숫자여야 합니다.") 

61 return float(value)

64def validate_layout(items, peak_photo_count, owned_photo_ids): 

65 """Return canonical JSON; overlap is allowed, leaving the room is not.""" 

66 policy = growth_policy(peak_photo_count) 

67 if not isinstance(items, list): 

68 raise LayoutValidationError("items는 배열이어야 합니다.") 

69 if len(items) > policy["limits"]["items"]: 

70 raise LayoutValidationError("현재 사진관의 장식 수 한도를 초과했습니다.") 

71 width, depth = policy["room"]["width"], policy["room"]["depth"] 

72 dimensions = {row[0]: row[2:] for row in _CATALOG} 

73 photos, seen, result, frames = set(owned_photo_ids), set(), [], 0 

74 for item in items: 

75 if not isinstance(item, dict) or set(item) != ITEM_KEYS: 

76 raise LayoutValidationError("장식 항목의 필드가 올바르지 않습니다.") 

77 item_id = item["id"] 

78 try: 

79 if not isinstance(item_id, str) or str(UUID(item_id)) != item_id.lower(): 

80 raise ValueError 

81 item_id = str(UUID(item_id)) 

82 except (ValueError, AttributeError, TypeError): 

83 raise LayoutValidationError("장식 ID는 하이픈을 포함한 UUID여야 합니다.") from None 

84 if item_id in seen: 

85 raise LayoutValidationError("장식 ID가 중복되었습니다.") 

86 seen.add(item_id) 

87 kind = item["kind"] 

88 if not isinstance(kind, str) or kind not in dimensions: 

89 raise LayoutValidationError("지원하지 않는 장식 종류입니다.") 

90 x, z = _number(item["x"], "x"), _number(item["z"], "z") 

91 rotation, scale = _number(item["rotation"], "rotation"), _number(item["scale"], "scale") 

92 if not MIN_SCALE <= scale <= MAX_SCALE: 

93 raise LayoutValidationError("장식 크기는 0.5부터 2 사이여야 합니다.") 

94 if not 0 <= rotation < 360: 

95 raise LayoutValidationError("회전은 0 이상 360 미만이어야 합니다.") 

96 photo_id, wall = item["photo_id"], item["wall"] 

97 item_width, item_depth, _height = dimensions[kind] 

98 if kind == "frame": 

99 frames += 1 

100 if frames > policy["limits"]["frames"]: 

101 raise LayoutValidationError("현재 사진관의 액자 수 한도를 초과했습니다.") 

102 if photo_id is not None and (type(photo_id) is not int or photo_id < 1 or photo_id not in photos): 

103 raise LayoutValidationError("액자에는 자신의 공개 사진만 연결할 수 있습니다.") 

104 if wall not in ("back", "left", "right"): 

105 raise LayoutValidationError("액자는 back, left, right 벽 중 하나에 붙여야 합니다.") 

106 expected_rotation = {"back": 0, "left": 90, "right": 270}[wall] 

107 if abs(rotation - expected_rotation) > 0.001: 

108 raise LayoutValidationError("액자의 회전은 선택한 벽의 방향과 같아야 합니다.") 

109 half_frame = item_width * scale / 2 

110 if wall == "back": 

111 in_bounds = abs(z - (-depth / 2 + WALL_INSET)) <= 0.001 and abs(x) + half_frame <= width / 2 - FLOOR_MARGIN 

112 elif wall == "left": 

113 in_bounds = abs(x - (-width / 2 + WALL_INSET)) <= 0.001 and abs(z) + half_frame <= depth / 2 - FLOOR_MARGIN 

114 else: 

115 in_bounds = abs(x - (width / 2 - WALL_INSET)) <= 0.001 and abs(z) + half_frame <= depth / 2 - FLOOR_MARGIN 

116 else: 

117 if wall is not None or photo_id is not None: 

118 raise LayoutValidationError("가구에는 벽이나 사진을 연결할 수 없습니다.") 

119 angle = radians(rotation) 

120 half_x = (abs(cos(angle)) * item_width + abs(sin(angle)) * item_depth) * scale / 2 

121 half_z = (abs(sin(angle)) * item_width + abs(cos(angle)) * item_depth) * scale / 2 

122 in_bounds = abs(x) + half_x <= width / 2 - FLOOR_MARGIN and abs(z) + half_z <= depth / 2 - FLOOR_MARGIN 

123 if not in_bounds: 

124 raise LayoutValidationError("장식이 사진관의 벽이나 바닥 범위를 벗어났습니다.") 

125 result.append({"id": item_id, "kind": kind, "x": x, "z": z, 

126 "rotation": rotation, "scale": scale, "photo_id": photo_id, "wall": wall}) 

127 return result

130def validate_gallery_metadata(values): 

131 if not isinstance(values, dict) or not values or set(values) - {"name", "description"}: 

132 raise LayoutValidationError("name 또는 description만 수정할 수 있습니다.") 

133 result = {} 

134 if "name" in values: 

135 name = values["name"] 

136 if not isinstance(name, str) or not name.strip() or len(name.strip()) > 80: 

137 raise LayoutValidationError("사진관 이름은 1부터 80자 사이여야 합니다.") 

138 result["name"] = name.strip() 

139 if "description" in values: 

140 description = values["description"]

142 raise LayoutValidationError("사진관 소개는 2000자 이하여야 합니다.") 

143 result["description"] = description
